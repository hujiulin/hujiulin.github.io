class Solution {
public:
    vector<vector<int> > levelOrder(TreeNode *root) {
        // IMPORTANT: Please reset any member data you declared, as
        // the same Solution instance will be reused for each test case.
        vector<vector<int>> ans;
        ans.clear();
        TreeNode ** seq = new TreeNode *[100000];
        TreeNode ** rseq = new TreeNode *[100000];

        int index = 0;
        seq[index++] = root;
        int top = index;
        while(true)
        {
            index = 0;
            vector<int> tmp;
            tmp.clear();
            int flag = 0;
            for(int i = 0;i < top;i++)
            {   
                if(seq[i] != NULL)
                {
                    if(seq[i]->left != NULL)
                        rseq[index++] = seq[i]->left;
                    if(seq[i]->right != NULL)
                        rseq[index++] = seq[i]->right;
                    tmp.push_back(seq[i]->val);
                    flag = 1;
                }
            }
            if(flag == 1)
                ans.push_back(tmp);
            else
                break;
            if(index == 0)
                break;
            for(int i = 0;i < index;i++)
            {
                seq[i] = rseq[i];
            }
            top = index;
        }
        return ans;
    }
};
